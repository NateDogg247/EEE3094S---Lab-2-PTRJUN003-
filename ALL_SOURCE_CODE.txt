"""Nominal closed-loop model, root locus, and step-response predictions."""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.signal import TransferFunction
from scipy.signal import step
from .plot_style import BLUE, ORANGE, GREEN, RED, exporter


def run(project_root, parameters):
    ROOT = project_root
    FIG = ROOT / "report/figures"
    RES = ROOT / "results"
    A, tau, ks, kp = (parameters.A, parameters.tau, parameters.ks, parameters.kp)
    b, kt = (parameters.b, parameters.kT)
    blue, orange, green, red = (BLUE, ORANGE, GREEN, RED)
    save = exporter(FIG)
    metrics = {}
    tt = np.linspace(
        0, parameters.end_s, round(parameters.end_s / parameters.interval_s) + 1
    )

    def closed(k=kp, drag=b, amp=5):
        _, yn = step(TransferFunction([kt * k * ks], [1, drag, kt * k * ks]), T=tt)
        return amp * yn

    def step_metrics(yy, amp=5):
        z = np.flatnonzero(np.abs(yy - amp) > 0.02 * abs(amp))
        return {
            "overshoot_percent": float(max(0, (yy.max() - amp) / amp * 100)),
            "peak_m": float(yy.max()),
            "peak_time_s": float(tt[np.argmax(yy)]),
            "settling_2percent_s": (
                float(tt[min(z[-1] + 1, len(tt) - 1)]) if len(z) else 0
            ),
            "terminal_error_m": float(amp - yy[-1]),
        }

    nom = closed()
    metrics["simulation_nominal"] = step_metrics(nom)
    wn = np.sqrt(kt * kp * ks)
    zeta = b / (2 * wn)
    wd = np.sqrt(wn**2 - (b / 2) ** 2)
    metrics.update(
        {
            "wn": wn,
            "zeta": zeta,
            "wd": wd,
            "sigma": b / 2,
            "overshoot_analytic_percent": 100
            * np.exp(-np.pi * zeta / np.sqrt(1 - zeta * zeta)),
            "peak_time_analytic_s": np.pi / wd,
            "settling_envelope_s": -np.log(0.02 * np.sqrt(1 - zeta * zeta)) / (b / 2),
        }
    )
    zmin = -np.log(0.3) / np.sqrt(np.pi**2 + np.log(0.3) ** 2)
    metrics["zeta_min_30percent"] = zmin
    metrics["kp_max_nominal"] = b * b / (4 * kt * ks * zmin * zmin)
    metrics["kp_max_joint_scalar10"] = (0.9 * b) ** 2 / (
        4 * kt * ks * zmin * zmin * 1.1
    )
    metrics["breakaway_gain"] = b * b / (4 * kt * ks)
    gg = np.linspace(0, 0.035, 1200)
    roots = np.array([np.roots([1, b, kt * k * ks]) for k in gg])
    fig, ax = plt.subplots(figsize=(7.1, 3.8))
    for i in range(2):
        ax.plot(roots[:, i].real, roots[:, i].imag, color=blue)
    ax.plot([0, -b], [0, 0], "x", ms=8, color="k", label="Open-loop poles")
    pp = np.roots([1, b, kt * kp * ks])
    ax.plot(pp.real, pp.imag, "o", color=orange, label="Kp = 0.01432")
    ax.axvline(-b / 2, color=green, ls=":", label="Re = −1/(2τ) = −0.06173")
    xx = np.linspace(-0.16, 0, 100)
    slope = np.sqrt(1 - zmin * zmin) / zmin
    ax.plot(xx, -xx * slope, "--", color="gray", label="30% overshoot boundary")
    ax.plot(xx, xx * slope, "--", color="gray")
    ax.axvline(-4 / 30, color=red, ls="--", label="30 s envelope target (unreachable)")
    ax.set(
        xlim=(-0.16, 0.012),
        ylim=(-0.25, 0.25),
        xlabel="Real part (s⁻¹)",
        ylabel="Imaginary part (s⁻¹)",
    )
    ax.legend(fontsize=9, loc="upper left")
    save(fig, "root_locus")
    fig, axs = plt.subplots(2, 1, figsize=(7.1, 3.8), sharex=True)
    axs[0].plot(tt, nom, color=blue, label="5 m step; nominal controller")
    axs[0].axhline(5, color="k", ls="--", label="Reference 5 m")
    axs[0].axhspan(4.9, 5.1, color=green, alpha=0.12, label="2% band")
    axs[0].plot(
        [metrics["simulation_nominal"]["peak_time_s"]], [nom.max()], "o", color=orange
    )
    axs[0].annotate(
        f"Peak {nom.max():.3f} m\nOvershoot {metrics['simulation_nominal']['overshoot_percent']:.2f}%",
        (metrics["simulation_nominal"]["peak_time_s"], nom.max()),
        xytext=(42, 6.0),
        arrowprops={"arrowstyle": "->"},
        fontsize=10,
    )
    axs[0].set_ylabel("Position (m)")
    axs[0].legend(fontsize=9, loc="lower right")
    cmd = 2.5 + kp * ks * (5 - nom)
    axs[1].plot(tt, cmd, color=orange)
    axs[1].axhline(2.5, color="gray", ls=":")
    axs[1].set(
        xlim=(0, 100), xlabel="Time from reference step (s)", ylabel="Command (V)"
    )
    fig.tight_layout()
    save(fig, "nominal_response")
    pd.DataFrame(
        {"time_s": tt, "reference_m": 5, "nominal_position_m": nom, "command_v": cmd}
    ).to_csv(RES / "nominal_simulation.csv", index=False)
    return metrics


def closed_response(time_s, parameters, gain=None, drag=None, amplitude=5):
    """Return a linear position step response with the sensor gain included."""
    gain = parameters.kp if gain is None else gain
    drag = parameters.b if drag is None else drag
    coefficient = parameters.kT * gain * parameters.ks
    _, response = step(
        TransferFunction([coefficient], [1, drag, coefficient]), T=time_s
    )
    return amplitude * response


def response_metrics(time_s, output_m, amplitude=5):
    """Measure overshoot and the final entry into the reference-based 2% band."""
    outside = np.flatnonzero(np.abs(output_m - amplitude) > 0.02 * abs(amplitude))
    return {
        "overshoot_percent": float(
            max(0, (output_m.max() - amplitude) / amplitude * 100)
        ),
        "peak_m": float(output_m.max()),
        "peak_time_s": float(time_s[np.argmax(output_m)]),
        "settling_2percent_s": (
            float(time_s[min(outside[-1] + 1, len(time_s) - 1)]) if len(outside) else 0
        ),
        "terminal_error_m": float(amplitude - output_m[-1]),
    }

"""Illustrative actuator-bias simulation; this is a prediction, not an experiment."""

import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp
from .plot_style import BLUE, ORANGE, GREEN, RED, exporter


def run(project_root, parameters):
    ROOT = project_root
    FIG = ROOT / "report/figures"
    RES = ROOT / "results"
    A, tau, ks, kp = (parameters.A, parameters.tau, parameters.ks, parameters.kp)
    b, kt = (parameters.b, parameters.kT)
    blue, orange, green, red = (BLUE, ORANGE, GREEN, RED)
    save = exporter(FIG)
    metrics = {}
    di = 0.002
    dtime = 70

    def simdist(t, z):
        yy, vv = z
        bias = di if t >= dtime else 0
        cmd = np.clip(2.5 + kp * ks * (5 - yy) + bias, 0, 5)
        return [vv, kt * (cmd - 2.5) - b * vv]

    s = solve_ivp(
        simdist,
        [0, 180],
        [0, 0],
        t_eval=np.linspace(0, 180, 3601),
        rtol=1e-09,
        atol=1e-10,
        max_step=0.02,
    )
    fig, ax = plt.subplots(figsize=(7.1, 3.1))
    ax.plot(s.t, s.y[0], color=blue)
    ax.axhline(5, color="k", ls="--")
    ax.axvline(dtime, color=orange, ls=":")
    ax.axhspan(4.6, 5.4, color=green, alpha=0.12)
    ax.set(xlabel="Time (s)", ylabel="Position (m)")
    ax.text(80, 5.85, "Illustrative +2 mV actuator bias at 70 s")
    save(fig, "disturbance_prediction")
    metrics["input_bias_limit_for_5m_8percent_v"] = 0.08 * 5 * kp * ks
    metrics["illustrative_bias_error_m"] = di / (kp * ks)
    return metrics

"""Session 1: differentiate position, identify the model, and validate it."""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.optimize import least_squares
from .plot_style import BLUE, ORANGE, GREEN, RED, exporter


def run(project_root, parameters):
    ROOT = project_root
    FIG = ROOT / "report/figures"
    RES = ROOT / "results"
    A, tau, ks, kp = (parameters.A, parameters.tau, parameters.ks, parameters.kp)
    b, kt = (parameters.b, parameters.kT)
    blue, orange, green, red = (BLUE, ORANGE, GREEN, RED)
    save = exporter(FIG)
    metrics = {}
    raw = pd.read_csv(ROOT / "data/session1/PTRJUN003_raw.csv")
    reset = np.flatnonzero(raw["Time(s)"].diff().to_numpy() < 0)
    start = int(reset[-1]) if len(reset) else 0
    d = raw.iloc[start:, :4].copy().reset_index(drop=True)
    t, u, sv, y = [d.iloc[:, i].to_numpy(float) for i in range(4)]
    if not np.all(np.diff(t) > 0):
        raise ValueError("Non-monotonic retained time")
    v = np.r_[np.nan, np.diff(y) / np.diff(t)]
    vmid = np.r_[np.nan, (t[:-1] + t[1:]) / 2]
    ss = float(np.mean(v[t >= 120]))
    Ass = ss / parameters.identification_step_v
    target = 0.632 * ss
    idx = np.flatnonzero(v >= target)[0]
    t63 = float(
        t[idx - 1]
        + (target - v[idx - 1]) / (v[idx] - v[idx - 1]) * (t[idx] - t[idx - 1])
    )
    sel = (t >= 17) & (t <= 60)
    fit = least_squares(
        lambda p: p[0] * (1 - np.exp(-(t[sel] - p[2]) / p[1])) - v[sel],
        [ss, 8.1, 15.9],
        bounds=([30, 3, 15.5], [40, 15, 16.2]),
        xtol=1e-12,
        ftol=1e-12,
        gtol=1e-12,
    )
    Vfit, taufit, t0fit = map(float, fit.x)
    cal = (sv > 0) & (sv < 9.9) & (y > 0)
    ksfit = float(y[cal] @ sv[cal] / (y[cal] @ y[cal]))

    def pred_vel(tt):
        return (
            A
            * parameters.identification_step_v
            * (1 - np.exp(-np.maximum(tt - t0fit, 0) / tau))
        )

    def pred_pos(tt):
        q = np.maximum(tt - t0fit, 0)
        return A * parameters.identification_step_v * (q - tau * (1 - np.exp(-q / tau)))

    vv = pred_vel(t)
    yy = pred_pos(t)
    train = (t >= 17) & (t <= 60)
    valid = t > 60

    def rmse(a, b):
        return float(np.sqrt(np.mean((a - b) ** 2)))

    metrics = {
        "retained_rows": len(d),
        "discarded_rows": start,
        "sample_interval_s": 0.1,
        "v_infinity_mps": ss,
        "A_from_tail": Ass,
        "v_63p2_mps": target,
        "t_63p2_absolute_s": t63,
        "tau_63p2_from_15p9_s": t63 - 15.9,
        "fit_v_infinity": Vfit,
        "fit_tau": taufit,
        "fit_step_time": t0fit,
        "ks_fit": ksfit,
        "calibration_points": int(cal.sum()),
        "velocity_rmse_fit_window": rmse(v[train], vv[train]),
        "velocity_rmse_holdout": rmse(v[valid], vv[valid]),
        "position_rmse_holdout": rmse(y[valid], yy[valid]),
        "nominal_A": A,
        "nominal_tau": tau,
        "nominal_ks": ks,
        "nominal_kp": kp,
        "b": b,
        "kT": kt,
    }
    d["Velocity(m/s)"] = v
    d["Velocity_time_midpoint(s)"] = vmid
    d["Model_velocity(m/s)"] = vv
    d["Model_position(m)"] = yy
    d.to_csv(RES / "session1_processed.csv", index=False)
    fig, axs = plt.subplots(3, 1, figsize=(7.1, 4.5), sharex=True)
    axs[0].plot(t, u, color=blue)
    axs[0].axhline(2.5, ls=":", color="gray")
    axs[0].set_ylabel("Input u (V)")
    axs[0].axvline(15.9, color=orange, ls="--")
    axs[0].text(25, 3.3, "Hover 2.5 V; final 5 V; Δu = 2.5 V")
    axs[1].plot(t, y, color=blue, label="Position channel (m)")
    axs[1].set_ylabel("Altitude y (m)")
    axs[2].plot(t, sv, color=green)
    axs[2].axhline(10, color=red, ls="--")
    axs[2].set_ylabel("Sensor (V)")
    axs[2].set_xlabel("Retained run time (s)")
    axs[2].text(45, 4.5, "Voltage clips at 10 V; metre channel keeps increasing")
    fig.tight_layout()
    save(fig, "identification_record")
    fig, ax = plt.subplots(figsize=(7.1, 3.5))
    q = t - 15.9
    ax.plot(q, v, color=blue, lw=1, label="Backward difference of position")
    ax.plot(q, vv, "--", color=orange, label="First-order model")
    ax.axhline(target, color=green, ls=":", label=f"63.2% = {target:.3f} m/s")
    ax.axvline(t63 - 15.9, color=green, ls=":")
    ax.axhline(ss, color="gray", ls=":")
    ax.set(
        xlim=(0, 60), ylim=(0, 39), xlabel="Time from step (s)", ylabel="Velocity (m/s)"
    )
    ax.text(12, 16, f"τ ≈ {t63 - 15.9:.2f} s; A ≈ {Ass:.4f} m/(s V)")
    ax.legend(loc="lower right", fontsize=10)
    save(fig, "velocity_identification")
    fig, axs = plt.subplots(1, 2, figsize=(7.1, 3.1))
    axs[0].plot(y[cal], sv[cal], "o", ms=3, color=blue, label="Unsaturated pairs")
    xx = np.array([0, 10.5])
    axs[0].plot(xx, ksfit * xx, color=orange, label=f"ks = {ksfit:.6f} V/m")
    axs[0].set(xlabel="Altitude (m)", ylabel="Sensor output (V)")
    axs[0].legend(fontsize=9)
    axs[1].plot(
        t[valid], v[valid] - vv[valid], color=blue, label="Held-out velocity residual"
    )
    axs[1].set(xlabel="Run time (s)", ylabel="Residual (m/s)")
    axs[1].legend(fontsize=9)
    fig.tight_layout()
    save(fig, "validation_sensor")
    fig, axs = plt.subplots(2, 1, figsize=(7.1, 3.8), sharex=True)
    axs[0].plot(t, y, color=blue, label="Recorded position")
    axs[0].plot(t, yy, "--", color=orange, label="Rounded model")
    axs[0].set_ylabel("Altitude (m)")
    axs[0].legend(fontsize=10)
    axs[1].plot(t, y - yy, color=green)
    axs[1].set(xlabel="Run time (s)", ylabel="Position residual (m)")
    fig.tight_layout()
    save(fig, "position_validation")
    return metrics

"""Load the same physical parameters used by MATLAB and the report."""

from dataclasses import dataclass
from pathlib import Path
import json


@dataclass(frozen=True)
class LabParameters:
    A: float
    tau: float
    ks: float
    kp: float
    hover_v: float
    identification_step_v: float
    reference_m: float
    end_s: float
    interval_s: float

    @property
    def b(self):
        return 1 / self.tau

    @property
    def kT(self):
        return self.A / self.tau

    @classmethod
    def load(cls, project_root: Path):
        values = json.loads((project_root / "config/lab_parameters.json").read_text())
        return cls(
            values["A"],
            values["tau_s"],
            values["sensor_gain_V_per_m"],
            values["controller_gain"],
            values["hover_command_V"],
            values["identification_step_V"],
            values["reference_step_m"],
            values["simulation_end_s"],
            values["simulation_interval_s"],
        )

"""Run the documented stages in order and save one consolidated result file."""

import json
from .parameters import LabParameters
from .plot_style import configure
from . import identification, controller, robustness, hardware, disturbance


def run_all(project_root):
    parameters = LabParameters.load(project_root)
    (project_root / "results").mkdir(exist_ok=True)
    configure()
    metrics = {}
    stages = [identification, controller, robustness, hardware, disturbance]
    for stage in stages:
        print("Running " + stage.__name__.split(".")[-1] + "...")
        metrics.update(stage.run(project_root, parameters))
    (project_root / "results/metrics.json").write_text(
        json.dumps(metrics, indent=2) + "\n"
    )
    return metrics

def exporter(directory):

"""Aerodynamic uncertainty, loaded circuit gain, and independent resistor corners."""

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.optimize import brentq
import itertools
from .plot_style import BLUE, ORANGE, GREEN, RED, exporter
from .controller import closed_response, response_metrics


def run(project_root, parameters):
    ROOT = project_root
    FIG = ROOT / "report/figures"
    RES = ROOT / "results"
    A, tau, ks, kp = (parameters.A, parameters.tau, parameters.ks, parameters.kp)
    b, kt = (parameters.b, parameters.kT)
    blue, orange, green, red = (BLUE, ORANGE, GREEN, RED)
    save = exporter(FIG)
    metrics = {}
    tt = np.linspace(
        0, parameters.end_s, round(parameters.end_s / parameters.interval_s) + 1
    )

    def closed(k=kp, drag=b, amp=5):
        return closed_response(tt, parameters, k, drag, amp)

    def step_metrics(y, amp=5):
        return response_metrics(tt, y, amp)

    fig, ax = plt.subplots(figsize=(7.1, 3.7))
    rows = []
    for factor in [0.9, 1, 1.1]:
        yc = closed(drag=factor * b)
        mm = step_metrics(yc)
        rows.append({"b_factor": factor, **mm})
        ax.plot(tt, yc, label=f"b × {factor:.1f}; Mp={mm['overshoot_percent']:.2f}%")
    ax.axhline(5, color="k", ls="--")
    ax.set(xlim=(0, 115), xlabel="Time (s)", ylabel="Position (m)")
    ax.legend(fontsize=10)
    save(fig, "aerodynamic_robustness")
    metrics["aerodynamic_corners"] = rows
    scalar = []
    for bf, kf in itertools.product([0.9, 1.1], [0.9, 1.1]):
        scalar.append(
            {
                "b_factor": bf,
                "kp_factor": kf,
                **step_metrics(closed(k=kp * kf, drag=b * bf)),
            }
        )
    metrics["scalar_corners"] = scalar
    Rnom = np.array(
        [100000, 1000, 100000, 1000, 4700, 10000, 100000, 100000, 100000, 100000], float
    )

    def bias_thevenin(w):
        r11, r12, r13, pot = (3900.0, 1200.0, 22000.0, 10000.0)
        pa, pb = (pot * w, pot * (1 - w))
        parallel = 1 / (1 / pot + 1 / r12)
        current = -15 / (r11 + parallel + r13)
        va = current * r11
        vb = current * (r11 + parallel)
        matrix = np.array(
            [
                [1 / r11 + 1 / r12 + 1 / pa, -1 / r12, -1 / pa],
                [-1 / r12, 1 / r13 + 1 / r12 + 1 / pb, -1 / pb],
                [-1 / pa, -1 / pb, 1 / pa + 1 / pb],
            ]
        )
        rth = float(np.linalg.solve(matrix, [0, 0, 1])[2])
        vth = va + w * (vb - va)
        return (vth, rth, -vth * 100000 / (100000 + rth))

    wpot = brentq(lambda w: bias_thevenin(w)[2] - 2.5, 0.01, 0.99)
    vth, rth, ubias = bias_thevenin(wpot)
    metrics["bias_network"] = {
        "w_fraction": wpot,
        "open_wiper_v": vth,
        "thevenin_ohm": rth,
        "loaded_hover_v": ubias,
        "loaded_kp": 0.01 * 1.47 * (1 + 100000 / (100000 + rth)) / 2,
    }
    corners = []
    for factors in itertools.product([0.9, 1.1], repeat=10):
        R = Rnom * np.array(factors)
        r1, r2, r3, r4, r5, r6, r7, r8, r9, r10 = R
        af = r4 / r3
        ar = (1 + af) * r2 / (r1 + r2)
        g2 = 1 + r5 / r6
        r = r10 / (r9 + rth)
        g3 = (1 + r) * r8 / (r7 + r8)
        K = af * g2 * g3
        ratio = ar / af
        for bf in [0.9, 1.1]:
            ze = bf * b / (2 * np.sqrt(kt * K * ks))
            os = 0 if ze >= 1 else 100 * np.exp(-np.pi * ze / np.sqrt(1 - ze * ze))
            corners.append([bf, K, ratio, abs(1 - ratio) * 100, os])
    co = pd.DataFrame(
        corners,
        columns=[
            "b_factor",
            "feedback_kp",
            "reference_feedback_ratio",
            "tracking_error_percent",
            "overshoot_percent",
        ],
    )
    co.to_csv(RES / "independent_resistor_corners.csv", index=False)
    metrics["resistor_corners"] = {
        "number": len(co),
        "kp_min": co.feedback_kp.min(),
        "kp_max": co.feedback_kp.max(),
        "worst_tracking_error_percent": co.tracking_error_percent.max(),
        "reference_ratio_min": co.reference_feedback_ratio.min(),
        "reference_ratio_max": co.reference_feedback_ratio.max(),
        "worst_overshoot_percent": co.overshoot_percent.max(),
        "nominal_loaded_prediction": 0.01 * 1.47 * (1 + 100000 / (100000 + rth)) / 2,
        "nominal_pot_fraction": wpot,
        "nominal_pot_thevenin_ohm": rth,
    }
    fig, axs = plt.subplots(1, 2, figsize=(7.1, 3.2))
    axs[0].hist(co.overshoot_percent, bins=28, color=blue)
    axs[0].axvline(30, color=red, ls="--")
    axs[0].set(xlabel="Overshoot (%)", ylabel="Corner count")
    axs[1].hist(co.tracking_error_percent, bins=28, color=orange)
    axs[1].axvline(8, color=red, ls="--")
    axs[1].set(xlabel="Tracking error after bias trim (%)", ylabel="Corner count")
    fig.tight_layout()
    save(fig, "component_robustness")
    newkp = 0.0068 * 1.47 * (1 + 100000 / (100000 + rth)) / 2
    metrics["recommended_kp"] = newkp
    metrics["recommended_worst_scalar_overshoot_percent"] = step_metrics(
        closed(k=newkp * 1.1, drag=0.9 * b)
    )["overshoot_percent"]
    return metrics

\subsection{Root-Locus Angle and Magnitude Checks}
The locus for variable $K_p$ uses $G_0(s)=k_sG_y(s)=(Ak_s/\tau)/[s(s+1/\tau)]$, where $Ak_s/\tau=1.638959$ s$^{-2}$. At the selected upper pole $s_d=-0.0617284+j0.1402123$,
\begin{align}
\angle s_d&=113.76147^\circ,\qquad
\angle(s_d+1/\tau)=66.23853^\circ,\\
\angle G_0(s_d)&=-113.76147^\circ-66.23853^\circ=-180^\circ,\\
K_p&=\frac1{|G_0(s_d)|}
=\frac{|s_d|\,|s_d+1/\tau|}{Ak_s/\tau}
=\frac{(0.153198845)^2}{1.63895852}=0.01432.
\end{align}
Both the negative-feedback angle criterion and magnitude criterion are satisfied. The quadratic roots independently give the same pole pair.

\subsection{Nominal Step and Settling Calculation}
Substituting into \Eref{eq:exactstep},
\begin{equation}
y(t)=5\left[1-e^{-0.061728395t}
\left(\cos(0.140212308t)+0.44025\sin(0.140212308t)\right)\right].
\end{equation}
The first peak occurs at $t_p=\pi/\omega_d=22.405969$ s. Since $\cos\pi=-1$ and $\sin\pi=0$,
\begin{equation}
y(t_p)=5[1+e^{-0.061728395(22.405969)}]=6.254019\unit{m},
\end{equation}
so $M_p=(6.254019-5)/5\times100=25.080375\%$.

The error envelope satisfies $|y-5|/5\le e^{-\sigma t}/\sqrt{1-\zeta^2}$. Solving this bound for 2\% gives
\begin{equation}
T_{\rm env}=-\frac{\ln(0.02\sqrt{1-0.4029299^2})}{0.061728395}=64.80975\unit{s}.
\end{equation}
Actual settling is found by evaluating the full response, locating the last sample for which $|y-5|>0.1$ m, and taking the next sample. A 0.005 s grid gives 54.89 s. This definition agrees with reference-based settling calculations \cite{stepinfo}.

Command headroom follows from $u=2.5+K_pk_s(5-y)$: the highest command is 2.566588 V at the step, and the lowest is $2.5-0.01432(0.93)(1.254019)=2.483299$ V at the altitude peak. Neither reaches the 0 or 5 V limit.

\subsection{Aerodynamic and Component Uncertainty}
With fixed $k_T$, the uncertain characteristic polynomial is $s^2+b's+k_TK_pk_s$. The worst scalar case has $b'=0.9b_0$ and $K_p'=1.1K_{p,0}$. Applying the required damping ratio gives
\begin{equation}\label{eq:jointbound}
K_{p,0}<\frac{(0.9b_0)^2}{4k_Tk_s\zeta_{\min}^2(1.1)}
=\frac{0.81}{1.1}(0.01815443)=0.01336826.
\end{equation}
At the supplied gain the worst-case damping is $(0.9/\sqrt{1.1})(0.4029299)=0.34575$, and \Eref{eq:overshoot} gives 31.42\% overshoot. This explains why passing drag-only uncertainty is insufficient for the joint case.

For independent loop-resistor tolerances, use the U1 coefficients and U2/U3 gains defined in \Sref{sec:hardware}. Assign each of $R_1$--$R_{10}$ either 0.9 or 1.1 times its nominal value, then use both drag extremes. There are $2^{10}\times2=2048$ cases. \subsection{Practical Gain and Ratio Correction}
The corner distributions in \Fref{fig:components} separate dynamic overshoot from DC ratio error.
\fig{component_robustness}{.92}{Loop-resistor corners. Dashed limits are 30\% overshoot and 8\% tracking error. The histograms show corner counts, not probabilities.}{fig:components}
Failure in this subset rejects a guarantee for all components. Bias-network and op-amp variations can add error. After matching the U1 ratios, a practical gain correction is to replace both 1 k$\Omega$ resistors with matched 680 $\Omega$ values:
\begin{align}
K_{p,\rm new}&=\frac{680}{100000}(1.47)(0.97427980)=0.009738901,\\
\zeta_{\rm worst}&=\frac{0.9b_0}{2\sqrt{k_T(1.1K_{p,\rm new})k_s}}\approx0.4193,\\
M_{p,\rm worst}&\approx23.44\%,\\
|\delta u_b|_{\max}&=0.08(5)(0.009738901)(0.93)=3.623\unit{mV}.
\end{align}
This scalar check assumes matched reference and feedback ratios; it is not a guarantee for independent resistor corners. Lower gain increases damping but tightens the hover-bias allowance. The board must be recalibrated and retested after any change.

\section{Circuit Gain and Bias Loading}\label{sec:hardware}\label{app:loading}
With the wiper unloaded, $R_P=P1\parallel R_{12}=10\parallel1.2=1.0714286$ k$\Omega$. The end-to-end current and pot-end voltages are
\begin{align}
I&=\frac{-15}{3.9+1.0714286+22}=-0.556143\unit{mA},\\
V_A&=I(3.9)=-2.168962\unit{V},\qquad V_B=I(3.9+1.0714286)=-2.764831\unit{V},\\
V_{\rm th}&=V_A+W(V_B-V_A).
\end{align}
Set the supply sources to zero and inject a 1 A test current at the wiper. The node-conductance equations for A, B and the wiper are
\begin{equation}
\begin{bmatrix}
G_{11}+G_{12}+G_t&-G_{12}&-G_t\\
-G_{12}&G_{13}+G_{12}+G_b&-G_b\\
-G_t&-G_b&G_t+G_b
\end{bmatrix}
\begin{bmatrix}V_A'\\V_B'\\V_w'\end{bmatrix}
=\begin{bmatrix}0\\0\\1\end{bmatrix},
\end{equation}
where $G_i=1/R_i$, $G_t=1/(WP1)$ and $G_b=1/[(1-W)P1]$. The test voltage $V_w'$ gives $R_{\rm th}$. Solving the hover equation
\begin{equation}
-\frac{R_{10}}{R_9+R_{\rm th}(W)}V_{\rm th}(W)=2.5
\end{equation}
gives $W=0.783080594$, $R_{\rm th}=5423.0014$ $\Omega$ and $V_{\rm th}=-2.6355750$ V. The loaded zero-error wiper is $-2.500$ V.

For U3, define $r=R_{10}/(R_9+R_{\rm th})=0.94855960$. Since the non-inverting divider gives $V_+=V_2/2$, virtual short and KCL give
\begin{align}
u&=(1+r)V_+-rV_{\rm th}=\frac{1+r}{2}V_2-rV_{\rm th},\\
g_3&=\frac{1+0.94855960}{2}=0.97427980,\\
K_p&=\left(\frac{1}{100}\right)\left(1+\frac{4.7}{10}\right)(0.97427980)=0.014321913.
\end{align}
The ideal buffered result 0.0147 differs by 2.57\%, while the selected model value 0.01432 differs from the loaded calculation by only 0.0134\%. These are circuit-model calculations; a measured input/output slope of the final board was not supplied.

function result = run_root_locus_design(P)
% Detailed root locus method and response checks. Requires Control System Toolbox.
%% 1. Adopt exactly the model and gain in the saved report
A = P.A;                 % Velocity DC gain, m/(s V)
tau = P.tau_s;                   % Velocity time constant, s
ks = P.sensor_gain_V_per_m;                   % Sensor gain, V/m
Kp = P.controller_gain;                % Report controller voltage-error gain, V/V
hoverV = P.hover_command_V;                % Equilibrium actuator command, V
reference_m = P.reference_step_m;             % Report nominal reference: 0 to 5 m
identificationDeltaU_V = P.identification_step_V; % Identification input: 2.5 to 5.0 V
% The identification actuator step and closed-loop position reference have
% different units. Do not substitute the 2.5 V identification step for 5 m.
b = 1/tau;
kT = A/tau;
s = tf('s');
Gpos = A/(s*(tau*s + 1));
G0 = ks*Gpos;                % Exclude Kp: rlocus varies this multiplier
L = Kp*G0;
T = feedback(L,1);          % y/reference; equivalent sensor-scaled loop
outDir = fullfile(P.project_root,'results','matlab','root_locus');
if ~exist(outDir,'dir'), mkdir(outDir); end
diary(fullfile(outDir,'root_locus_calculations.txt'));
fprintf('REPORT MODEL: A=%.4f, tau=%.1f, ks=%.2f, Kp=%.5f\n',A,tau,ks,Kp);
fprintf('Identification actuator increment: %.1f V; position reference: %.1f m\n',...
    identificationDeltaU_V,reference_m);
disp('G0(s), with sensor gain but without the variable Kp:'); disp(G0);

%% 2. Open-loop poles, real-axis section, centroid and breakaway
% Poles: 0 and -1/tau. There are no finite zeros.
% Real-axis locus: [-1/tau,0]. Two asymptotes: +90 and -90 degrees.
% Kp(s) = -s*(tau*s+1)/(A*ks).
% dKp/ds = -(2*tau*s+1)/(A*ks) = 0 at sb = -1/(2*tau).
openPoles = pole(G0);
openZeros = zero(G0);
sb = -1/(2*tau);
Kbreak = 1/(4*tau*A*ks);
fprintf('\nOpen-loop poles:\n'); disp(openPoles);
fprintf('Number of finite zeros: %d\n',numel(openZeros));
fprintf('Centroid and breakaway pole: %.9f 1/s\n',sb);
fprintf('Breakaway gain: %.9f\n',Kbreak);

%% 3. Convert the 30%% overshoot limit into a damping/gain limit
MpLimit = 0.30;              % Fraction, not percentage
zetaMin = -log(MpLimit)/sqrt(pi^2 + log(MpLimit)^2);
% Characteristic polynomial: tau*s^2+s+A*ks*Kp = 0.
% wn = sqrt(A*ks*Kp/tau); zeta = 1/(2*sqrt(tau*A*ks*Kp)).
KmaxNominal = 1/(4*tau*A*ks*zetaMin^2);
% For aerodynamic uncertainty, vary b while kT stays fixed.
% Scalar gain tolerance models overall gain only, not resistor mismatch.
KmaxJointScalar = (0.9*b)^2/(4*kT*ks*zetaMin^2*1.1);
fprintf('\nMinimum damping ratio for <30%% overshoot: %.9f\n',zetaMin);
fprintf('Nominal overshoot limit: 0 < Kp < %.9f\n',KmaxNominal);
fprintf('Joint +/-10%% drag/gain limit: Kp < %.9f\n',KmaxJointScalar);

%% 4. Calculate the selected closed-loop pole pair
q = A*ks*Kp;
selectedPoles = roots([tau 1 q]);
wn = sqrt(q/tau);
sigma = 1/(2*tau);
zeta = sigma/wn;
wd = sqrt(wn^2-sigma^2);
MpAnalytic = 100*exp(-pi*zeta/sqrt(1-zeta^2));
tp = pi/wd;
TsApprox = 4/sigma;          % Standard 2%% settling approximation
TsEnvelope = -log(0.02*sqrt(1-zeta^2))/sigma;
fprintf('\nCharacteristic polynomial: %.1f*s^2+s+%.9f = 0\n',tau,q);
fprintf('Selected poles:\n'); disp(selectedPoles);
fprintf('wn=%.9f rad/s; zeta=%.9f; wd=%.9f rad/s\n',wn,zeta,wd);
fprintf('Analytic overshoot=%.6f%%; peak time=%.6f s\n',MpAnalytic,tp);
fprintf('4/sigma approximation=%.6f s; 2%% envelope bound=%.6f s\n',...
    TsApprox,TsEnvelope);

%% 5. Check the root locus angle and magnitude conditions
sd = complex(-sigma,wd);     % Upper selected pole
theta1 = atan2d(imag(sd),real(sd));
theta2 = atan2d(imag(sd),real(sd)+b);
GatPole = evalfr(G0,sd);
angleGdeg = angle(GatPole)*180/pi;
KfromMagnitude = 1/abs(GatPole);
KfromDistances = abs(sd)*abs(sd+b)/(kT*ks);
charResidual = abs(tau*sd^2+sd+q);
fprintf('\nAngle from pole 0 = %.6f degrees\n',theta1);
fprintf('Angle from pole -1/tau = %.6f degrees\n',theta2);
fprintf('Pole angle sum=%.6f degrees; angle G0(sd)=%.6f degrees\n',...
    theta1+theta2,angleGdeg);
fprintf('Magnitude gain 1/|G0(sd)| = %.9f\n',KfromMagnitude);
fprintf('Pole-distance gain = %.9f; characteristic residual = %.3g\n',...
    KfromDistances,charResidual);

%% 6. Plot the root locus and the specification boundaries
gainVector = linspace(0,0.04,2001);
r = rlocus(G0,gainVector);   % Each column corresponds to one gain
angelBlue = [143 201 230]/255;
darkBlue = [57 123 156]/255;
f1 = figure('Color','w');
plot(real(r(1,:)),imag(r(1,:)),'Color',darkBlue,'LineWidth',1.8);
hold on;
plot(real(r(2,:)),imag(r(2,:)),'Color',darkBlue,'LineWidth',1.8,...
    'HandleVisibility','off');
plot(real(openPoles),imag(openPoles),'kx','MarkerSize',10,'LineWidth',2);
plot(real(selectedPoles),imag(selectedPoles),'o','Color',[0.85 0.35 0.12],...
    'MarkerFaceColor',[0.85 0.35 0.12],'MarkerSize',7);
plot(sb,0,'ks','MarkerFaceColor',angelBlue,'MarkerSize',7);
xline(sb,':','Fixed complex-pole real part','Color',[0.2 0.5 0.3]);
x = linspace(-0.18,0,200);
slope = sqrt(1-zetaMin^2)/zetaMin;
plot(x,-x*slope,'--','Color',[0.5 0.5 0.5]);
plot(x,x*slope,'--','Color',[0.5 0.5 0.5],'HandleVisibility','off');
TsTarget = 30;
sigmaTarget = 4/TsTarget;
xline(-sigmaTarget,'--','30 s approximate settling boundary',...
    'Color',[0.7 0.2 0.2]);
grid on; xlabel('Real part (1/s)'); ylabel('Imaginary part (1/s)');
title(sprintf('Kitticopter root locus: Kp = %.5f',Kp));
xlim([-0.18 0.015]); ylim([-0.30 0.30]);
exportgraphics(f1,fullfile(outDir,'root_locus.png'),'Resolution',200);
exportgraphics(f1,fullfile(outDir,'root_locus.pdf'),'ContentType','vector');
fprintf('\n30 s design line: Re(s) <= %.9f\n',-sigmaTarget);
fprintf('Available complex-pole real part: %.9f\n',sb);
fprintf(['The standard 30 s decay-rate target is outside this P-only locus.\n'...
    'This line uses Ts approximately 4/sigma; the exact response is checked next.\n']);

%% 7. Verify the step response and report-compatible settling time
t = (0:0.005:160)';
y = reference_m*step(T,t);
info = stepinfo(y,t,reference_m,'SettlingTimeThreshold',0.02);
% Also reproduce the report's sampled final-entry definition explicitly.
outside = find(abs(y-reference_m)>0.02*reference_m);
if isempty(outside)
    TsSampled = 0;
elseif outside(end)==numel(t)
    TsSampled = NaN;          % Not settled within the simulated interval
else
    TsSampled = t(outside(end)+1);
end
u = hoverV+Kp*ks*(reference_m-y);
steadyGain = dcgain(T);
fprintf('\nMATLAB stepinfo results:\n'); disp(info);
fprintf('Sampled last-entry 2%% settling time=%.6f s\n',TsSampled);
fprintf('DC tracking gain=%.9f; ideal step tracking error=%.9f%%\n',...
    steadyGain,100*abs(1-steadyGain));
fprintf('Command range=%.6f to %.6f V\n',min(u),max(u));
f2 = figure('Color','w');
subplot(2,1,1);
plot(t,y,'Color',darkBlue,'LineWidth',1.5); hold on;
yline(reference_m,'k--','Reference');
yline(0.98*reference_m,':','2% lower bound');
yline(1.02*reference_m,':','2% upper bound');
grid on; ylabel('Position (m)'); title('Report nominal 0 to 5 m response');
subplot(2,1,2);
plot(t,u,'Color',[0.85 0.35 0.12],'LineWidth',1.5); hold on;
yline(hoverV,':','Hover command'); grid on;
xlabel('Time from reference step (s)'); ylabel('Actuator command (V)');
exportgraphics(f2,fullfile(outDir,'nominal_response.png'),'Resolution',200);
writetable(table(t,y,u,'VariableNames',{'time_s','position_m','command_V'}),...
    fullfile(outDir,'nominal_response.csv'));

%% 8. Aerodynamic constant uncertainty: b times 0.9, 1.0 and 1.1
% With kT fixed, tau'=1/b' and A'=kT/b': do not vary tau alone.
f3 = figure('Color','w'); hold on; grid on;
for bf = [0.9 1.0 1.1]
    Tb = tf(kT*Kp*ks,[1 bf*b kT*Kp*ks]);
    yb = reference_m*step(Tb,t);
    ib = stepinfo(yb,t,reference_m,'SettlingTimeThreshold',0.02);
    plot(t,yb,'LineWidth',1.4,'DisplayName',sprintf('b x %.1f',bf));
    fprintf('b x %.1f: overshoot %.6f%%; stepinfo settling %.6f s\n',...
        bf,ib.Overshoot,ib.SettlingTime);
end
yline(reference_m,'k--','HandleVisibility','off'); legend('Location','best');
xlabel('Time (s)'); ylabel('Position (m)'); title('Aerodynamic uncertainty');
exportgraphics(f3,fullfile(outDir,'aerodynamic_uncertainty.png'),'Resolution',200);

%% 9. Joint scalar gain and drag uncertainty: four corners
% Independent circuit-resistor tolerance is NOT covered by these four cases.
% The original analysis/analyse.py evaluates 2,048 resistor/drag corners.
cornerResults = zeros(4,4);
row = 0;
for bf = [0.9 1.1]
    for kf = [0.9 1.1]
        row = row+1;
        Tc = tf(kT*(kf*Kp)*ks,[1 bf*b kT*(kf*Kp)*ks]);
        yc = reference_m*step(Tc,t);
        ic = stepinfo(yc,t,reference_m,'SettlingTimeThreshold',0.02);
        cornerResults(row,:) = [bf kf ic.Overshoot ic.SettlingTime];
    end
end
cornerTable = array2table(cornerResults,'VariableNames',...
    {'drag_factor','gain_factor','overshoot_percent','settling_time_s'});
disp('Joint scalar uncertainty results:'); disp(cornerTable);
writetable(cornerTable,fullfile(outDir,'joint_scalar_corners.csv'));
fprintf('Largest scalar-corner overshoot: %.6f%%\n',max(cornerResults(:,3)));
fprintf('Detailed independent component tolerance: run analysis/analyse.py.\n');
diary off;
fprintf('\nFigures, tables and calculation log saved in:\n%s\n',outDir);

result = struct("gain",Kp,"poles",selectedPoles,"damping_ratio",zeta,...
    "overshoot_percent",MpAnalytic,"settling_time_s",TsSampled);
end

function result = identify_session1(P)
% Identify gain, time constant and sensor gain from the original Session 1 CSV.
% This is a MATLAB equivalent; report estimates are generated by Python.
raw = readmatrix(fullfile(P.project_root,'data','session1','PTRJUN003_raw.csv'));
resetIndex = find(diff(raw(:,1)) < 0,1,'last');
if ~isempty(resetIndex), raw = raw(resetIndex+1:end,:); end
t = raw(:,1); u = raw(:,2); sensor = raw(:,3); y = raw(:,4);
v = [NaN; diff(y)./diff(t)];
vInf = mean(v(t >= 120));
A = vInf/P.identification_step_V;
target = 0.632*vInf;
i = find(v >= target,1,'first');
t63 = t(i-1)+(target-v(i-1))/(v(i)-v(i-1))*(t(i)-t(i-1));
tau = t63-15.9;
validSensor = sensor>0 & sensor<9.9 & y>0;
ks = sum(y(validSensor).*sensor(validSensor))/sum(y(validSensor).^2);
result = struct('velocity_final_m_per_s',vInf,'A',A,'time_constant_s',tau,...
    'sensor_gain_V_per_m',ks,'retained_rows',numel(t));
disp('Session 1 estimates:'); disp(result);
figure('Color','w'); subplot(2,1,1); plot(t,u); grid on;
ylabel('Input (V)'); title('Retained Session 1 run');
subplot(2,1,2); plot(t-15.9,v); hold on;
modelV = P.A*P.identification_step_V*(1-exp(-max(t-15.9,0)/P.tau_s));
plot(t-15.9,modelV,'--'); grid on;
xlabel('Time from step (s)'); ylabel('Velocity (m/s)');
legend('Position backward difference','Adopted model');
end

function verify_nominal_controller(P)
% Original independent report check, using the shared parameters.
A=P.A; tau=P.tau_s; ks=P.sensor_gain_V_per_m; Kp=P.controller_gain;
b=1/tau; kT=A/tau;
s=tf('s'); Gy=A/(s*(tau*s+1));
L=Kp*ks*Gy; T=feedback(L,1);
t=(0:0.005:160)'; y=5*step(T,t);
wn=sqrt(A*Kp*ks/tau); zeta=1/(2*sqrt(tau*A*Kp*ks));
wd=wn*sqrt(1-zeta^2);
fprintf('Poles:'); disp(pole(T));
fprintf('zeta=%.7f, wn=%.7f, wd=%.7f\n',zeta,wn,wd);
fprintf('Analytic Mp=%.6f%%, tp=%.6f s\n',100*exp(-pi*zeta/sqrt(1-zeta^2)),pi/wd);
info=stepinfo(y,t,5,'SettlingTimeThreshold',0.02);
disp(info);
u=2.5+Kp*ks*(5-y);
assert(all(u>=0 & u<=5),'Nominal command is outside 0-5 V');
figure; rlocus(ks*Gy); grid on; hold on;
plot(real(pole(T)),imag(pole(T)),'ro','MarkerFaceColor','r');
title('Proportional root locus; sensor gain included');
xline(-1/(2*tau),'--','Fixed complex-pole real part');
figure; subplot(2,1,1);plot(t,y,t,5*ones(size(t)),'--');
yline(4.9,':');yline(5.1,':'); ylabel('Position (m)');grid on;
subplot(2,1,2);plot(t,u);ylabel('Command (V)');xlabel('Time (s)');grid on;
figure; hold on;
for bf=[0.9 1 1.1]
 Tb=tf(kT*Kp*ks,[1 bf*b kT*Kp*ks]);
 yb=5*step(Tb,t);ib=stepinfo(yb,t,5,'SettlingTimeThreshold',0.02);
 fprintf('b x %.1f: Mp %.5f%%, Ts %.5f s\n',bf,ib.Overshoot,ib.SettlingTime);
 plot(t,yb,'DisplayName',sprintf('b x %.1f',bf));
end
legend;grid on; xlabel('Time (s)');ylabel('Position (m)');
fprintf('Joint scalar worst Mp: ');
Tw=tf(kT*(1.1*Kp)*ks,[1 .9*b kT*(1.1*Kp)*ks]);
iw=stepinfo(5*step(Tw,t),t,5,'SettlingTimeThreshold',0.02);disp(iw.Overshoot);

end

\subsection{Bias Loading and Stage Gain Calculations}\label{app:loading}
With the wiper unloaded, $R_P=P1\parallel R_{12}=10\parallel1.2=1.0714286$ k$\Omega$. The end-to-end current and pot-end voltages are
\begin{align}
I&=\frac{-15}{3.9+1.0714286+22}=-0.556143\unit{mA},\\
V_A&=I(3.9)=-2.168962\unit{V},\qquad V_B=I(3.9+1.0714286)=-2.764831\unit{V},\\
V_{\rm th}&=V_A+W(V_B-V_A).
\end{align}
Set the supply sources to zero and inject a 1 A test current at the wiper. The node-conductance equations for A, B and the wiper are
\begin{equation}
\begin{bmatrix}
G_{11}+G_{12}+G_t&-G_{12}&-G_t\\
-G_{12}&G_{13}+G_{12}+G_b&-G_b\\
-G_t&-G_b&G_t+G_b
\end{bmatrix}
\begin{bmatrix}V_A'\\V_B'\\V_w'\end{bmatrix}
=\begin{bmatrix}0\\0\\1\end{bmatrix},
\end{equation}
where $G_i=1/R_i$, $G_t=1/(WP1)$ and $G_b=1/[(1-W)P1]$. The test voltage $V_w'$ gives $R_{\rm th}$. Solving the hover equation
\begin{equation}
-\frac{R_{10}}{R_9+R_{\rm th}(W)}V_{\rm th}(W)=2.5
\end{equation}
gives $W=0.783080594$, $R_{\rm th}=5423.0014$ $\Omega$ and $V_{\rm th}=-2.6355750$ V. The loaded zero-error wiper is $-2.500$ V.

For U3, define $r=R_{10}/(R_9+R_{\rm th})=0.94855960$. Since the non-inverting divider gives $V_+=V_2/2$, virtual short and KCL give
\begin{align}
u&=(1+r)V_+-rV_{\rm th}=\frac{1+r}{2}V_2-rV_{\rm th},\\
g_3&=\frac{1+0.94855960}{2}=0.97427980,\\
K_p&=\left(\frac{1}{100}\right)\left(1+\frac{4.7}{10}\right)(0.97427980)=0.014321913.
\end{align}
The ideal buffered result 0.0147 differs by 2.57\%, while the selected model value 0.01432 differs from the loaded calculation by only 0.0134\%. These are circuit-model calculations; a measured input/output slope of the final board was not supplied.

\subsection{Root-Locus Angle and Magnitude Checks}
The locus for variable $K_p$ uses $G_0(s)=k_sG_y(s)=(Ak_s/\tau)/[s(s+1/\tau)]$, where $Ak_s/\tau=1.638959$ s$^{-2}$. At the selected upper pole $s_d=-0.0617284+j0.1402123$,
\begin{align}
\angle s_d&=113.76147^\circ,\qquad
\angle(s_d+1/\tau)=66.23853^\circ,\\
\angle G_0(s_d)&=-113.76147^\circ-66.23853^\circ=-180^\circ,\\
K_p&=\frac1{|G_0(s_d)|}
=\frac{|s_d|\,|s_d+1/\tau|}{Ak_s/\tau}
=\frac{(0.153198845)^2}{1.63895852}=0.01432.
\end{align}
Both the negative-feedback angle criterion and magnitude criterion are satisfied. The quadratic roots independently give the same pole pair.

\subsection{Nominal Step and Settling Calculation}
Substituting into \Eref{eq:exactstep},
\begin{equation}
y(t)=5\left[1-e^{-0.061728395t}
\left(\cos(0.140212308t)+0.44025\sin(0.140212308t)\right)\right].
\end{equation}
The first peak occurs at $t_p=\pi/\omega_d=22.405969$ s. Since $\cos\pi=-1$ and $\sin\pi=0$,
\begin{equation}
y(t_p)=5[1+e^{-0.061728395(22.405969)}]=6.254019\unit{m},
\end{equation}
so $M_p=(6.254019-5)/5\times100=25.080375\%$.

The error envelope satisfies $|y-5|/5\le e^{-\sigma t}/\sqrt{1-\zeta^2}$. Solving this bound for 2\% gives
\begin{equation}
T_{\rm env}=-\frac{\ln(0.02\sqrt{1-0.4029299^2})}{0.061728395}=64.80975\unit{s}.
\end{equation}
Actual settling is found by evaluating the full response, locating the last sample for which $|y-5|>0.1$ m, and taking the next sample. A 0.005 s grid gives 54.89 s. This definition agrees with reference-based settling calculations \cite{stepinfo}.

Command headroom follows from $u=2.5+K_pk_s(5-y)$: the highest command is 2.566588 V at the step, and the lowest is $2.5-0.01432(0.93)(1.254019)=2.483299$ V at the altitude peak. Neither reaches the 0 or 5 V limit.

\subsection{Aerodynamic and Component Uncertainty}
With fixed $k_T$, the uncertain characteristic polynomial is $s^2+b's+k_TK_pk_s$. The worst scalar case has $b'=0.9b_0$ and $K_p'=1.1K_{p,0}$. Applying the required damping ratio gives
\begin{equation}\label{eq:jointbound}
K_{p,0}<\frac{(0.9b_0)^2}{4k_Tk_s\zeta_{\min}^2(1.1)}
=\frac{0.81}{1.1}(0.01815443)=0.01336826.
\end{equation}
At the supplied gain the worst-case damping is $(0.9/\sqrt{1.1})(0.4029299)=0.34575$, and \Eref{eq:overshoot} gives 31.42\% overshoot. This explains why passing drag-only uncertainty is insufficient for the joint case.

For independent loop-resistor tolerances, use the U1 coefficients and U2/U3 gains defined in \Sref{sec:hardware}. Assign each of $R_1$--$R_{10}$ either 0.9 or 1.1 times its nominal value, then use both drag extremes. There are $2^{10}\times2=2048$ cases. The bias network is held at its nominal Thevenin resistance and zero-error output is recalibrated in each case. \Tref{tab:corners} reports this optimistic subset.

\subsection{Practical Gain and Ratio Correction}
The corner distributions in \Fref{fig:components} separate dynamic overshoot from DC ratio error.
\fig{component_robustness}{.92}{Loop-resistor corners. Dashed limits are 30\% overshoot and 8\% tracking error. The histograms show corner counts, not probabilities.}{fig:components}
Failure in this subset rejects a guarantee for all components. Bias-network and op-amp variations can add error. After matching the U1 ratios, a practical gain correction is to replace both 1 k$\Omega$ resistors with matched 680 $\Omega$ values:
\begin{align}
K_{p,\rm new}&=\frac{680}{100000}(1.47)(0.97427980)=0.009738901,\\
\zeta_{\rm worst}&=\frac{0.9b_0}{2\sqrt{k_T(1.1K_{p,\rm new})k_s}}\approx0.4193,\\
M_{p,\rm worst}&\approx23.44\%,\\
|\delta u_b|_{\max}&=0.08(5)(0.009738901)(0.93)=3.623\unit{mV}.
\end{align}
This scalar check assumes matched reference and feedback ratios; it is not a guarantee for independent resistor corners. Lower gain increases damping but tightens the hover-bias allowance. The board must be recalibrated and retested after any change.

\subsection{Experimental Metrics and Uncertainty}

\begin{align}
t&=\frac{x-602}{1414-602}(120),\qquad h=\frac{540-p_y}{540-60}(20),\\
\Delta t&=0.147783\unit{s/pixel},\qquad\Delta h=0.041667\unit{m/pixel}.
\end{align}
The extracted peak is 6.229167 m at 22.019704 s. With the displayed reference, terminal height and zero starting altitude,
\begin{align}
e_f&=5.033-4.897=0.136\unit{m},\\
\eta_{\rm track}&=100(1-0.136/5.033)=97.297834\%,\\
M_{p,\rm ref}&=100(6.229167-5.033)/5.033=23.766475\%,\\
\delta M_{p,\rm ref}&\approx100(0.041667)/5.033=0.82787\text{ percentage points}.

This practical overshoot metric uses the commanded reference. \section{Introduction}\label{sec:intro}
\subsection{Overview}
The Kitticopter laboratory combines a physical analogue controller with a simulated vertical-motion plant through a data-acquisition (DAQ) interface. Session 1 identifies the plant from an applied voltage step. Session 2 uses that model to design, implement and test an altitude controller on Veroboard. The modelling, identification, design and evaluation below follow this experimental sequence \cite{template}.

\subsection{Aim}
The aim is to regulate altitude using proportional control while satisfying the four requirements in the 2026 laboratory brief \cite{lab}. \Tref{tab:requirements} translates these requirements into quantities that can be checked against the measured and simulated responses.
\begin{table}[H]\centering\small
\caption{Laboratory requirements and their evaluation measures.}\label{tab:requirements}
\begin{tabularx}{\linewidth}{p{37mm}X}\toprule\thead
Requirement & Evaluation measure\\\midrule
Tracking accuracy $>92\%$ & $100(1-|x-y_\infty|/|x|)>92\%$ for a nonzero step reference.\\
Overshoot $<30\%$ & $M_p=100(y_{\max}-y_\infty)/(y_\infty-y_0)<30\%$.\\

\section{System Modelling}\label{sec:model}
\subsection{Methodology}
The laboratory permits a unit-mass model with velocity-proportional aerodynamic drag and constant gravity \cite{lab}. Upward altitude is positive. Newton's second law is applied first, the actuator is linearised around hover, and the Laplace transform then gives a first-order velocity model. Integrating velocity produces the position model used for control.

\subsubsection{Dynamic Model}\label{sec:dynamic}
Newton's second law, with mass normalised to one, gives
\begin{equation}\label{eq:forces}
\ddot y=f_t-b\dot y-g+d_a.
\end{equation}
At hover $f_{t,h}=g$. The local actuator relationship is $f_t=g+k_T(u-u_h)$; substituting this into \Eref{eq:forces} gives
\begin{equation}\label{eq:dynamic}
\ddot y+b\dot y=k_T\delta u+d_a,\qquad \delta u=u-u_h.
\end{equation}
The hover operating point does not imply $k_T=g/u_h$: the slope must be obtained from the step response. With zero initial conditions and disturbances set to zero,
\begin{align}
G_{vf}(s)&=\frac{V(s)}{F_t(s)}=\frac{1}{s+b},\\
G_v(s)&=\frac{V(s)}{\Delta U(s)}=\frac{k_T}{s+b}=\frac{A}{\tau s+1},\quad
\tau=\frac1b,\quad A=\frac{k_T}{b},\label{eq:velocityplant}\\
G_y(s)&=\frac{Y(s)}{\Delta U(s)}=\frac{G_v(s)}s=\frac{A}{s(\tau s+1)}.\label{eq:plant}
\end{align}
Thus the velocity response is first order and the position plant is second order. The integrator explains why a nonzero voltage deviation produces an altitude ramp after the velocity transient.

The reference is converted to voltage as $r_v=k_sx$, while the sensor gives $y_v=k_sy$. The implemented control law is
\begin{equation}\label{eq:law}
u=\operatorname{sat}_{[0,5]}\{2.5+K_p(r_v-y_v)\}.
\end{equation}

\subsubsection{Closed-Loop Representation}
In the unsaturated region, $\Delta U=K_pk_s(X-Y)$. Substituting into \Eref{eq:plant} and collecting the output terms gives
\begin{equation}
Y=G_yK_pk_s(X-Y)\ \Rightarrow\ (1+G_yK_pk_s)Y=G_yK_pk_sX.
\end{equation}
Define $q=AK_pk_s$ and $D(s)=\tau s^2+s+q$. The reference and position-error transfer functions are
\begin{align}
L(s)&=K_pk_s\frac{A}{s(\tau s+1)},\\
\frac{Y(s)}{X(s)}&=\frac{L(s)}{1+L(s)}=\frac{q}{D(s)},\label{eq:tracking}\\
\frac{E_y(s)}{X(s)}&=1-\frac{Y(s)}{X(s)}=\frac{\tau s^2+s}{D(s)},\qquad E_y=X-Y.
\end{align}
An additional acceleration enters \Eref{eq:dynamic} directly. Therefore
\begin{equation}\label{eq:disturbance}
\frac{Y(s)}{D_a(s)}=\frac{\tau}{D(s)},\qquad
\frac{Y(s)}{G(s)}=-\frac{\tau}{D(s)}\quad\text{for uncompensated gravity}.
\end{equation}
The 2.5 V bias compensates nominal gravity in the actual implementation.

\subsubsection{Steady-State Properties}
The loop is Type 1 because $L(s)$ contains one pole at the origin. For $K_p>0$, the characteristic polynomial $s^2+bs+k_TK_pk_s$ has positive coefficients and is stable. The final-value theorem gives zero step-reference error. For a ramp $x(t)=a_rt$, $K_v=\lim_{s\to0}sL(s)=AK_pk_s$, so $e_\infty=a_r/K_v$ is finite; a parabolic reference has infinite steady-state error.

A constant disturbance behaves differently from a reference step. Applying the final-value theorem to \Eref{eq:disturbance} gives
\begin{equation}\label{eq:bias}
y_\infty-x_\infty=\frac{d_0}{k_TK_pk_s}
=\frac{\delta u_b}{K_pk_s}\quad\text{when }d_0=k_T\delta u_b.
\end{equation}
For a 5 m reference and $K_p=0.01432$, the 8\% error allowance requires
\begin{equation}
|\delta u_b|<0.08(5)(0.01432)(0.93)=0.005327\unit{V}.
\end{equation}
Hover bias must consequently be calibrated to about 5.3 mV. The brief does not specify disturbance amplitude or injection point; disturbance compliance is evaluated for explicitly defined cases in \Aref{app:calculations}.

\subsection{Data Processing}
\subsubsection{Velocity Computation}
The metre channel is differentiated using a backward difference:
\begin{equation}\label{eq:diff}
v_i=\frac{y_i-y_{i-1}}{t_i-t_{i-1}},\qquad t_{v,i}=\frac{t_i+t_{i-1}}2.
\end{equation}
For example, the 16.9 and 17.0 s positions are 2.256 and 2.703 m, giving $v=(2.703-2.256)/0.1=4.47$ m/s at midpoint 16.95 s. No smoothing is applied. The stored speed column agrees to CSV rounding. Differentiating the clipped voltage channel would incorrectly produce zero velocity; the metre channel avoids this error.

\subsubsection{Estimation of Gain and Time Constant}
The 404 velocity samples with endpoint time $t\ge120$ s give $v_\infty=35.68748$ m/s. With $v_0=0$,
\begin{equation}\label{eq:gainid}
A=\frac{v_\infty-v_0}{\Delta u}=\frac{35.68748-0}{5.0-2.5}
=14.27499\unit{m/(s\,V)}.
\end{equation}
The first-order step response is $v(t)=v_\infty[1-e^{-(t-t_0)/\tau}]$. At one time constant, $1-e^{-1}\approx0.632$, so $v_{63}=0.632(35.68748)=22.55448$ m/s. The measured crossing lies between 24.0 and 24.1 s:
\begin{align}
t_{63}&=24.0+\frac{22.55448-22.53}{22.69-22.53}(24.1-24.0)=24.01530\unit{s},\\
\tau_{63}&=t_{63}-t_0\approx24.01530-15.9=8.11530\unit{s}.
\end{align}
A least-squares exponential fit over 17--60 s gives $V_\infty=35.68747$ m/s, $\tau=8.10393$ s and $t_0=15.91584$ s. Endpoint-labelled differences are used in that fit; the midpoint shift is absorbed in $t_0$, leaving $\tau$ unchanged.

\subsubsection{Sensor Calibration (V/m)}
Using 22 nonzero pairs with $0<y_v<9.9$ V, the zero-intercept least-squares coefficient is
\begin{equation}\label{eq:sensor}
k_s=\frac{\sum_i y_i y_{v,i}}{\sum_i y_i^2}=0.9300026\unit{V/m}\approx0.93\unit{V/m}.
\end{equation}
The single-point check at 17.0 s is $2.513/2.703=0.92971$ V/m. The regression uses multiple unsaturated pairs rather than the clipped 10 V tail. Full processing equations and selected sample calculations are in \Aref{app:calculations}.

\section{Controller Design}\label{sec:design}
\subsection{Design Process}
\subsubsection{Plant for Design}
The design uses \Eref{eq:identified}, $k_s=0.93$ V/m and a 2.5 V hover bias. Its characteristic equation is
\begin{equation}\label{eq:characteristic}
1+K_pk_sG_y(s)=0\quad\Rightarrow\quad8.1s^2+s+14.2748(0.93)K_p=0.
\end{equation}

\subsubsection{Design Specifications}
For a standard underdamped second-order response,
\begin{equation}\label{eq:overshoot}
M_p=100e^{-\pi\zeta/\sqrt{1-\zeta^2}},\quad
M_p<30\%\Rightarrow\zeta>\frac{-\ln0.3}{\sqrt{\pi^2+\ln^20.3}}=0.357857.
\end{equation}
Matching the monic form of \Eref{eq:characteristic} to $s^2+2\zeta\omega_ns+\omega_n^2$ yields
\begin{equation}\label{eq:second}
\omega_n=\sqrt{\frac{AK_pk_s}{\tau}},\qquad
\zeta=\frac{1}{2\sqrt{\tau AK_pk_s}},\qquad
\sigma=\zeta\omega_n=\frac{1}{2\tau}.
\end{equation}
Therefore $K_p<1/(4\tau Ak_s\zeta_{\min}^2)=0.0181544$ for nominal overshoot compliance. Type 1 tracking is ideal only after bias and reference/sensor scaling are calibrated. A practical nominal validation target is a 2\% band within 60 s; for drag-only uncertainty, 75 s is reasonable. These are reporting targets added to the four official requirements, not guarantees for uncalibrated component corners.

\subsubsection{Gain Selection (Root Locus)}
The open-loop poles are $0$ and $-1/\tau=-0.1234568$; there are no finite zeros. The real-axis locus lies between the poles, and its two asymptotes have angles $\pm90^\circ$ about centroid $-1/(2\tau)$. Differentiating $K_p(s)=-s(\tau s+1)/(Ak_s)$ gives
\begin{equation}
\frac{dK_p}{ds}=0\Rightarrow s_b=-\frac{1}{2\tau}=-0.0617284,\quad
K_{p,b}=\frac{1}{4\tau Ak_s}=0.00232489.
\end{equation}
The supplied gain $K_p=0.01432$ is above breakaway and below the nominal overshoot limit. Since the complex branches have fixed real part $-0.0617284$, a 30 s decay-envelope target ($\sigma\approx4/30$) cannot be imposed by proportional gain alone. Detailed angle and magnitude checks are in \Aref{app:calculations}; the \RepoLink{matlab/design}{root-locus code folder} reproduces the method.
\page

\subsubsection{Design Summary}
Substituting $K_p=0.01432$ gives $q=14.2748(0.01432)(0.93)=0.1901061$ and
\begin{equation}
G_{cl}(s)=\frac{0.1901061}{8.1s^2+s+0.1901061}
=\frac{0.02346989}{s^2+0.1234568s+0.02346989}.
\end{equation}
The roots and response quantities in \Tref{tab:design} are calculated from \Eref{eq:second} and checked against \Fref{fig:locus}.

The 2,048 loop-resistor/drag corners yield up to 48.79\% tracking error and 38.95\% overshoot (\Fref{fig:components}). Calculations and practical corrections follow in \Aref{app:calculations} and \Sref{sec:evaluation}.

\subsection{Test Procedure}
The documented DAQ mapping is DAC-1 for reference, DAC-0 for sensor feedback, ADC-1 for command and ADC-2 for sensor monitoring. Equal reference and sensor inputs are used to trim the zero-error command to 2.500 V. The retained test display then shows an altitude response from near zero to a reference of 5.033 m. At 86.10 s, the recorded reference is 5.033 m and height is 4.897 m. The display reports 0.137 m tracking error, 2.498 V command and 4.553 V sensor output. The computed error $5.033-4.897=0.136$ m differs from the tracking readout by only 0.001 m. The sensor check $4.553/4.896=0.92994$ V/m agrees with Session 1.

\section{Conclusion and Recommendations}\label{sec:conclusion}
\subsection{Conclusion}
Identification supports $G_y(s)=14.2748/[s(8.1s+1)]$ and $k_s=0.93$ V/m. Bias loading gives $K_p\approx0.01432$. Nominal tracking and overshoot pass; component mismatch and joint uncertainty prevent full robustness.

\subsection{Recommendations}\label{sec:recommendations}
\begin{enumerate}
\item Match/trim U1 ratios to below 1\% mismatch and verify both signal slopes.
\item Use matched 680 $\Omega$ U1 resistors: proposed gain 0.00974, joint overshoot 23.44\%.
\item Recalibrate to 2.500 V; the revised 5 m bias allowance is 3.623 mV.
\item Retain repeated raw steps and defined disturbance tests to assess robustness.
\end{enumerate}

"""Verify the user's page-count rule and all report hyperlinks."""

from pathlib import Path
from urllib.parse import urlparse
import json
from pypdf import PdfReader

ROOT = Path(__file__).resolve().parents[1]
PDF = ROOT / "report/PTRJUN003_Lab2_Report.pdf"
REPO = "https://github.com/NateDogg247/EEE3094S---Lab-2-PTRJUN003-"


def check():
    reader = PdfReader(PDF)
    labels = reader.page_labels
    reference_index = labels.index("R1")
    intro_index = reference_index - 14
    main_labels = labels[intro_index:reference_index]
    assert main_labels == [str(i) for i in range(1, 15)], main_labels
    assert "Introduction" in reader.pages[intro_index].extract_text()
    assert "Final Remarks" in reader.pages[reference_index - 1].extract_text()
    assert labels[reference_index + 1] == "A1"
    assert "List of Figures" in reader.pages[intro_index - 2].extract_text()
    assert "List of Tables" in reader.pages[intro_index - 1].extract_text()
    internal_count = github_count = 0
    destinations = reader.named_destinations
    github_urls = set()
    for page in reader.pages:
        for item in page.get("/Annots", []):
            annotation = item.get_object()
            action = annotation.get("/A", {})
            if action.get("/S") == "/URI":
                uri = str(action["/URI"])
                if uri.startswith(REPO):
                    assert urlparse(uri).scheme == "https"
                    github_count += 1
                    github_urls.add(uri)
                    for prefix in (REPO + "/tree/main/", REPO + "/blob/main/"):
                        if uri.startswith(prefix):
                            assert (ROOT / uri[len(prefix) :]).exists(), uri
            if action.get("/S") == "/GoTo":
                destination = action["/D"]
                if isinstance(destination, str):
                    assert destination in destinations, destination
                internal_count += 1
            assert action.get("/S") != "/Launch"
    assert github_count > 10
    assert internal_count > 50
    assert not any("??" in page.extract_text() for page in reader.pages)
    result = {
        "main_pages": 14,
        "excluded_title_pages": 1,
        "front_matter_pages": intro_index - 1,
        "reference_pages": 1,
        "appendix_pages": len(reader.pages) - reference_index - 1,
        "total_pages": len(reader.pages),
        "internal_links": internal_count,
        "github_links": github_count,
        "github_urls": sorted(github_urls),
    }
    (ROOT / "results/report_checks.json").write_text(
        json.dumps(result, indent=2) + "\n"
    )
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    check()

from pathlib import Path
from urllib.parse import urlparse

    "url", help="Actual HTTPS repository URL, or local for internal PDF links"
)

a = p.parse_args()
root = Path(__file__).resolve().parents[1]
if not re.fullmatch(r"[A-Za-z0-9._/-]+", a.branch):
    p.error("Unsafe branch name")
if a.url == "local":
    text = (root / "scripts/repository_internal.tex").read_text()
else:
    url = a.url.rstrip("/").removesuffix(".git")
    u = urlparse(url)
    if (
        u.scheme != "https"
        or u.netloc not in ["github.com", "gitlab.com"]
        or u.query
        or u.fragment
    ):
        p.error("Use a plain HTTPS github.com or gitlab.com repository URL")
    if not re.fullmatch(r"/[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+", u.path):
        p.error("Use the repository root URL")
    tree = "/-/tree/" if u.netloc == "gitlab.com" else "/tree/"
    text = (
        r"\newcommand{\RepoRoot}{"
        + url
        + "}\n"
        + r"\newcommand{\RepoBranch}{"
        + a.branch
        + "}\n"
        + r"\newcommand{\RepoLink}[2]{\href{\RepoRoot"
        + tree
        + r"\RepoBranch/#1}{#2}}"
        + "\n"
        + r"\newcommand{\RepoFile}[2]{\href{\RepoRoot/blob/\RepoBranch/#1}{#2}}"
        + "\n"
        + r"\newcommand{\RepoStatus}{The supporting-folder links open the published repository on the configured branch.}"

    )
(root / "report/repository.tex").write_text(text)
